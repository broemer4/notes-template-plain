\subsection{Group Theory}

% Definition
\begin{fdefinition}[Group]
    A group $(G,\cdot)$ consists of a set $G$ and an operation $\cdot: G \times G \rightarrow G$, satisfying the following properties:
    \begin{enumerate}
        \item Associativity: $(a \cdot b) \cdot c = a \cdot (b \cdot c)$ for all $a,b,c \in G$
        \item Identity element: There exists an element $e \in G$ such that $a \cdot e = e \cdot a = a$ for all $a \in G$
        \item Inverse element: For each $a \in G$, there exists an element $a^{-1} \in G$ such that $a \cdot a^{-1} = a^{-1} \cdot a = e$
    \end{enumerate}
\end{fdefinition}

% Theorem
\begin{ftheorem}[Lagrange's Theorem]
    Let $G$ be a finite group and $H$ be a subgroup of $G$. Then the order of $H$ divides the order of $G$.
\end{ftheorem}

% Proof
\begin{proof}
    Let $G$ be a finite group and $H$ be a subgroup of $G$. We can partition the elements of $G$ into equivalence classes induced by $H$: $G = \bigcup_{g \in G} gH$. Note that every element of $gH$ has the same order as any other element of $gH$. Therefore, $|gH| = |H|$ for all $g \in G$. Since the equivalence classes are disjoint, we have $|G| = |H| \cdot n$ for some positive integer $n$, which implies that $|H|$ divides $|G|$.
\end{proof}

% Corollary
\begin{fcorollary}
    Let $G$ be a finite group and $a \in G$. Then $a^{|G|} = e$, where $e$ is the identity element of $G$.
\end{fcorollary}

% Proof
\begin{proof}
    Let $G$ be a finite group and $a \in G$. Consider the cyclic subgroup generated by $a$: $\langle a \rangle = {a^n \mid n \in \mathbb{Z}}$. Since $\langle a \rangle$ is a subgroup of $G$, Lagrange's Theorem implies that $|\langle a \rangle|$ divides $|G|$. Thus, $a^{|G|} = (a^{|\langle a \rangle|})^{k} = e^k = e$, where $k = |G|/|\langle a \rangle|$.
\end{proof}
